local function makeDraggable(guiObject)
	local dragging, dragInput, dragStart, startPos
	guiObject.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = guiObject.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	guiObject.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	uis.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

makeDraggable(mainFrame)

local hexColors = {
	"#2020f5",
	"#d5f520",
	"#f52080",
	"#42e6f5"
}

task.spawn(function()
	local index = 1
	while true do
		local nextIndex = index + 1
		if nextIndex > #hexColors then nextIndex = 1 end
		
		local tweenInfo = TweenInfo.new(2.5, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut)
		local tween = ts:Create(titleLabel, tweenInfo, {TextColor3 = Color3.fromHex(hexColors[nextIndex])})
		tween:Play()
		tween.Completed:Wait()
		
		index = nextIndex
	end
end)

fireBtn.MouseButton1Click:Connect(function()
	local charName = nameInput.Text ~= "" and nameInput.Text or "Luffy"
	local charRarity = rarityInput.Text ~= "" and rarityInput.Text or "Rare"
	local charMutation = mutationInput.Text ~= "" and mutationInput.Text or "Normal"
	
	local eventsFolder = rep:FindFirstChild("Events")
	if eventsFolder and eventsFolder:FindFirstChild("RewardEscape") then
		local args = {
			[1] = charName,
			[2] = charMutation,
			[3] = charRarity,
			n = 3
		}
		eventsFolder.RewardEscape:FireServer(unpack(args, 1, args.n or #args))
		print("Fired event locally with custom character params")
	else
		warn("Events.RewardEscape not found in ReplicatedStorage")
	end
end)
